# ── Harnesses lab ─────────────────────────────────────────────────────────────
# Append this block to your shared Caddyfile (e.g. /root/caddy-router/Caddyfile
# if you're following this repo's layout). This repo does not run Caddy itself
# — see the "Project layout" note in README.md.
#
# Current deployment: ONE unified wildcard block handles every hostname under
# *.lab.example.com — both the 11 base harnesses (claude, aider, opencode,
# crush, gptme, goose, plandex, qwencode, codex, pi, droid) AND dynamic
# `<harness>-<slug>` sessions (see README's "Multiple simultaneous sessions").
# Every session of one harness type — base AND every dynamic slug — shares
# that ONE harness-<type> container and its one /workspace, each on its own
# port. Caddy doesn't need to know the port: it reverse-proxies to whatever
# `X-Harness-Upstream` auth/server.py's /verify response returns (copied into
# the request by the harness_auth snippet's copy_headers).
#
# Prerequisites:
#   1. DNS:  add an `*.lab.example.com  A  <VPS_IP>` wildcard record at your
#            DNS provider. Without this, `claude-foo.lab.example.com` (or even
#            plain `claude.lab.example.com`) won't resolve no matter what
#            Caddy does.
#   2. Global block: on-demand TLS needs an `ask` endpoint so Caddy only
#            issues certs for hostnames the auth service actually recognizes
#            (otherwise anyone requesting `*.lab.example.com` triggers a cert
#            issuance attempt). Add this to the GLOBAL { … } block at the top
#            of your Caddyfile:
#              on_demand_tls {
#                  ask http://harnesses-auth:8080/ask
#              }
#
# Auth flow per visit:
#   1. Visitor opens  https://<harness>.lab.example.com/?token=<jwt>
#      (or https://<harness>-<slug>.lab.example.com/?token=<jwt> for an extra
#      session — any existing token for that harness type works).
#   2. @hastoken handler rewrites to /issue and proxies to harnesses-auth.
#      Auth validates the token, sets an HttpOnly cookie scoped to
#      .lab.example.com (one cookie unlocks every subdomain), and 302s to /.
#   3. Subsequent requests hit forward_auth, which calls auth
#      /verify?harness=<name>. Auth validates the cookie, ensures the target
#      container is running (cold-start blocks 3-7s), and for a
#      `<harness>-<slug>` hostname docker-execs an extra tmux+ttyd session
#      into that SAME base container if one isn't already there.
#   4. On 200, Caddy proxies to the harness/port named in the response's
#      X-Harness-Upstream header.
#
# Reload after editing:
#   docker exec caddy-router caddy reload --config /etc/caddy/Caddyfile
#
# IMPORTANT: do not add a separate static `<name>.lab.example.com { ... }`
# block alongside the wildcard block below — Caddy's automatic HTTPS treats
# an exact-hostname site and a wildcard site covering it as a policy
# conflict, which breaks cert resolution for every hostname under the
# wildcard. Every hostname under *.lab.example.com belongs to the one
# wildcard block, always.

(harness_auth) {
    # Explicit route { } so directives execute in source order — without it,
    # Caddy's auto-order puts forward_auth BEFORE the @hastoken handler and
    # the `?token=…` first-visit flow never fires.
    route {
        @hastoken query token=*
        handle @hastoken {
            rewrite * /issue
            reverse_proxy harnesses-auth:8080
        }

        # /pin and /unpin go back to the auth service (toggle retention for
        # the dynamic session corresponding to the calling subdomain).
        @pinpath path /pin /unpin
        handle @pinpath {
            forward_auth harnesses-auth:8080 {
                uri /verify?harness={args[0]}
                copy_headers Cookie
            }
            reverse_proxy harnesses-auth:8080
        }

        forward_auth harnesses-auth:8080 {
            uri /verify?harness={args[0]}
            # X-Harness-Upstream carries back the actual "container:port" to
            # proxy to (auth allocates a port dynamically — multiple sessions
            # of one harness type share a single container on different
            # ports, so the hostname alone doesn't determine the target).
            copy_headers Cookie X-Harness-Upstream
            # ttyd uses WebSockets; without stripping the upgrade headers
            # uvicorn rejects the /verify GET with 400 Bad Request and Caddy
            # then 403s the WS upgrade to the user's browser.
            header_up -Connection
            header_up -Upgrade
            header_up -Sec-Websocket-Version
            header_up -Sec-Websocket-Key
            header_up -Sec-Websocket-Protocol
            header_up -Sec-Websocket-Extensions
        }
    }
}

# {http.request.host.labels.3} is the leftmost label of a 4-label
# *.lab.example.com host (labels are indexed from the right: for
# "claude-blog.lab.example.com" -> 0=com, 1=example, 2=lab, 3=claude-blog),
# i.e. the full harness-or-instance slug — passed to harness_auth as the
# /verify?harness= query param so auth/server.py knows which session this is.
*.lab.example.com {
    tls {
        on_demand
    }
    import harness_auth {http.request.host.labels.3}
    reverse_proxy {http.request.header.X-Harness-Upstream} {
        flush_interval -1
        header_up Host {host}
    }
}
