# ── Harnesses lab ─────────────────────────────────────────────────────────────
# Append this block to /root/caddy-router/Caddyfile.
#
# DNS:  wildcard A record  *.lab.casava.space  →  this VPS
#
# Auth flow per harness:
#   1. Visitor opens  https://<harness>.lab.casava.space/?token=<JWT>
#   2. @hastoken handler rewrites to /issue and proxies to harnesses-auth.
#      Auth validates the JWT, sets a host-scoped HttpOnly cookie, redirects
#      back to /  (so the URL bar stops showing the secret).
#   3. Subsequent requests fall through to `forward_auth` which calls
#      auth /verify?harness=<name>.  Auth validates the cookie AND ensures
#      the container is running (cold-start blocks 3-7s).
#   4. On 200, Caddy proxies to the harness container by service name.
#
# Reload after editing:
#   docker exec caddy-router caddy reload --config /etc/caddy/Caddyfile

(harness_auth) {
    @hastoken query token=*
    handle @hastoken {
        rewrite * /issue
        reverse_proxy harnesses-auth:8080 {
            header_up X-Forwarded-Host {host}
        }
    }

    forward_auth harnesses-auth:8080 {
        uri /verify?harness={args[0]}
        copy_headers Cookie
    }
}

claude.lab.casava.space {
    import harness_auth claude
    reverse_proxy harness-claude:7681 {
        flush_interval -1
        header_up Host {host}
    }
}
