# ── Harnesses lab ─────────────────────────────────────────────────────────────
# Append this block to /root/caddy-router/Caddyfile.
#
# Current deployment: each harness has its own A record + dedicated site block.
# Single-instance only.
#
# Multi-instance via wildcard subdomains is implemented in the auth-service
# (it creates per-slug containers + volumes on demand, with retention).  The
# Caddy layer to activate it is at the bottom of this file under "## OPTIONAL".
# Two prerequisites before enabling that:
#
#   1. DNS:  add an `*.lab.casava.space  A  <VPS_IP>` wildcard record at your
#            DNS provider.  Without this, `claude-foo.lab.casava.space` won't
#            resolve no matter what Caddy does.
#   2. Caddy on-demand TLS: HTTP-01 + wildcard hostnames can be tricky.  Test
#            after enabling that openhands.lab + kilocode.lab still serve
#            their existing certs from /data/caddy without re-issuance.
#
# Auth flow per visit (current single-instance layout):
#   1. Visitor opens  https://<harness>.lab.casava.space/?token=<JWT_SECRET-or-jwt>
#   2. @hastoken handler rewrites to /issue and proxies to harnesses-auth.
#      Auth validates the token, sets an HttpOnly cookie scoped to .lab.casava.space
#      (one cookie unlocks every subdomain), and 302s to /.
#   3. Subsequent requests hit forward_auth which calls auth /verify?harness=<name>.
#      Auth validates the cookie AND ensures the container is running
#      (cold-start blocks 3-7s).
#   4. On 200, Caddy proxies to the harness container by service name.
#
# Reload after editing:
#   docker exec caddy-router caddy reload --config /etc/caddy/Caddyfile

(harness_auth) {
    # Explicit route { } so directives execute in source order — without it,
    # Caddy's auto-order puts forward_auth BEFORE the @hastoken handler and
    # the `?token=…` first-visit flow never fires.
    route {
        @hastoken query token=*
        handle @hastoken {
            rewrite * /issue
            reverse_proxy harnesses-auth:8080
        }

        # /pin and /unpin go back to the auth service (toggle retention for
        # the instance corresponding to the calling subdomain).  Only meaningful
        # once the wildcard block at the bottom of this file is enabled — for
        # single-instance subdomains the auth-service returns a friendly 400.
        @pinpath path /pin /unpin
        handle @pinpath {
            forward_auth harnesses-auth:8080 {
                uri /verify?harness={args[0]}
                copy_headers Cookie
            }
            reverse_proxy harnesses-auth:8080
        }

        forward_auth harnesses-auth:8080 {
            uri /verify?harness={args[0]}
            copy_headers Cookie
            # ttyd uses WebSockets; without stripping the upgrade headers
            # uvicorn rejects the /verify GET with 400 Bad Request and Caddy
            # then 403s the WS upgrade to the user's browser.
            header_up -Connection
            header_up -Upgrade
            header_up -Sec-Websocket-Version
            header_up -Sec-Websocket-Key
            header_up -Sec-Websocket-Protocol
            header_up -Sec-Websocket-Extensions
        }
    }
}
